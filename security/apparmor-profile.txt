# AppArmor profile for StackCodeSy
# This profile restricts system calls and file access for enhanced security
#
# Installation:
#   1. Copy to /etc/apparmor.d/stackcodesy
#   2. Run: apparmor_parser -r /etc/apparmor.d/stackcodesy
#   3. Add to docker-compose.yml:
#      security_opt:
#        - apparmor=stackcodesy

#include <tunables/global>

profile stackcodesy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # Allow network access (TCP/UDP)
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # Deny raw sockets (prevents packet sniffing)
  deny network raw,
  deny network packet,

  # File system access - restrictive
  # Allow read access to necessary system files
  /etc/passwd r,
  /etc/group r,
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/nsswitch.conf r,
  /etc/localtime r,

  # Allow workspace directory (read/write)
  /workspace/** rw,
  /stackcodesy/** rw,
  /home/stackcodesy/.stackcodesy/** rw,
  /tmp/** rw,

  # Node.js and application binaries
  /usr/bin/node ix,
  /usr/local/bin/node ix,
  /stackcodesy/scripts/** ix,

  # Deny access to sensitive system directories
  deny /proc/sys/kernel/** wl,
  deny /sys/kernel/** wl,
  deny /boot/** rwlx,
  deny /root/** rwlx,
  deny /etc/shadow* rwlx,
  deny /etc/sudoers* rwlx,

  # Deny dangerous capabilities
  deny capability sys_admin,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability sys_boot,
  deny capability sys_time,
  deny capability net_admin,

  # Allow only necessary capabilities
  capability chown,
  capability dac_override,
  capability setgid,
  capability setuid,
  capability fowner,
  capability kill,

  # Deny mount operations
  deny mount,
  deny umount,
  deny remount,

  # Deny ptrace (prevents debugging other processes)
  deny ptrace,

  # Allow signals within profile
  signal (send receive) peer=stackcodesy,

  # Deny execution of interpreters from /tmp
  deny /tmp/** x,

  # Deny access to other users' home directories
  deny /home/*/** rwlx,
  /home/stackcodesy/** rw,

  # Library paths
  /lib/** mr,
  /lib64/** mr,
  /usr/lib/** mr,

  # Proc filesystem (limited access)
  @{PROC}/@{pid}/stat r,
  @{PROC}/@{pid}/status r,
  @{PROC}/@{pid}/fd/ r,
  @{PROC}/sys/kernel/osrelease r,
  @{PROC}/sys/kernel/ostype r,
  deny @{PROC}/kcore rwlx,
  deny @{PROC}/kmem rwlx,
  deny @{PROC}/mem rwlx,

  # Deny device access
  deny /dev/mem rwlx,
  deny /dev/kmem rwlx,
  deny /dev/port rwlx,

  # Allow terminal devices
  /dev/pts/* rw,
  /dev/tty rw,
  /dev/null rw,
  /dev/zero r,
  /dev/random r,
  /dev/urandom r,
}
